## Note - You will need to format your sdcard on another machine.
Create a single partition and format it as an ext4 filesystem.

Example here:

```bash
wipefs -af /dev/mmcblk0
echo "/dev/mmcblk0p1 : start=1, type=83" > /tmp/part.info
sfdisk --no-reread -f /dev/mmcblk0 < /tmp/part.info
rm /tmp/part.info
mkfs.ext4 /dev/mmcblk0p1
```

## Make Nginx listen only on the main interface.  For a 3 node cluster HA Proxy will be listening on a new interface

```bash
sed -i "s|listen 80|listen 10.11.12.1:80|g" /etc/nginx/conf.d/gl.conf
sed -i "s|listen 443|listen 10.11.12.1:443|g" /etc/nginx/conf.d/gl.conf
sed -i "/listen \[::\]:80/d" /etc/nginx/conf.d/gl.conf
sed -i "/listen \[::\]:443/d" /etc/nginx/conf.d/gl.conf
/etc/init.d/nginx restart
```

## Set up DNS

```bash
export EDGE_ROUTER=10.11.12.1
export EDGE_NETWORK=10.11.12.0
export EDGE_NETMASK=255.255.255.0
export EDGE_CIDR=24
export LAB_DOMAIN=my.awesome.lab
export ARPA=12.11.10

mkdir -p /usr/local/bind

cat << EOF > /usr/local/bind/named.conf
acl "trusted" {
 ${EDGE_NETWORK}/${EDGE_CIDR};
 127.0.0.1;
};

options {
 listen-on port 53 { 127.0.0.1; ${EDGE_ROUTER}; };
 
 directory  "/tmp";
 allow-query     { trusted; };
 allow-recursion { trusted; };
 recursion yes;

 forwarders {
   8.8.8.8;
   8.8.4.4;
  };
  auth-nxdomain no;

 include "/etc/bind/named-rndc.conf";
 include "/tmp/bind/named.conf.local";

};

zone "${LAB_DOMAIN}" {
    type master;
    file "/usr/local/bind/db.${LAB_DOMAIN}"; # zone file path
};

zone "${ARPA}.in-addr.arpa" {
    type master;
    file "/usr/local/bind/db.${ARPA}";
};

zone "localhost" {
    type master;
    file "/etc/bind/db.local";
};

zone "127.in-addr.arpa" {
    type master;
    file "/etc/bind/db.127";
};

zone "0.in-addr.arpa" {
    type master;
    file "/etc/bind/db.0";
};

zone "255.in-addr.arpa" {
    type master;
    file "/etc/bind/db.255";
};
EOF

cat << EOF > /usr/local/bind/db.${LAB_DOMAIN}
@       IN      SOA     router.${LAB_DOMAIN}. admin.${LAB_DOMAIN}. (
             3          ; Serial
             604800     ; Refresh
              86400     ; Retry
            2419200     ; Expire
             604800 )   ; Negative Cache TTL
;
; name servers - NS records
    IN      NS     router.${LAB_DOMAIN}.

; name servers - A records
router.${LAB_DOMAIN}.         IN      A      ${EDGE_ROUTER}
EOF

cat << EOF > /usr/local/bind/db.${ARPA}
@       IN      SOA     router.${LAB_DOMAIN}. admin.${LAB_DOMAIN}. (
                            3         ; Serial
                        604800         ; Refresh
                        86400         ; Retry
                        2419200         ; Expire
                        604800 )       ; Negative Cache TTL

; name servers - NS records
    IN      NS      router.${LAB_DOMAIN}.

; PTR Records
1    IN      PTR     router.${LAB_DOMAIN}.
EOF
```

# Copyright (C) 2009-2010 OpenWrt.org

START=99
STOP=80

HAPROXY_CONFIG="/etc/haproxy-${CLUSTER_NAME}.cfg"
HAPROXY_PID="/var/run/haproxy-${CLUSTER_NAME}.pid"

start() {
	service_start \$HAPROXY_BIN -q -D -f "\$HAPROXY_CONFIG" -p "\$HAPROXY_PID"
}

stop() {
	kill -9 \$(cat \$HAPROXY_PID)
	service_stop \$HAPROXY_BIN
}

reload() {
	\$HAPROXY_BIN -D -q -f \$HAPROXY_CONFIG -p \$HAPROXY_PID -sf \$(cat \$HAPROXY_PID)

        \$HAPROXY_BIN -c -q -V -f \$HAPROXY_CONFIG
